// =====================================================
//                  CONVEYOR BELT

// =====================================================
//                  FISH FEEDER

// =====================================================
//                  ULTRASONIC
// =====================================================

#define TRIG_PIN 22
#define ECHO_PIN 23


// =====================================================
//                  TURBIDITY
// =====================================================

#define TURBIDITY_PIN 32


// =====================================================
//                  SENSOR VARIABLES
// =====================================================

long duration;
int distance;
int wastePercent;
int turbidityPercent;


// =====================================================
//                  MOTOR FUNCTIONS
// =====================================================

void forwardMotor() {

  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);

  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
}


void backwardMotor() {

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
}


void leftMotor() {

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);

  digitalWrite(IN3, HIGH);
  digitalWrite(IN4, LOW);
}


void rightMotor() {

  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, HIGH);
}


void stopMotor() {

  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);

  digitalWrite(IN3, LOW);
  digitalWrite(IN4, LOW);
}


// =====================================================
//                  BELT FUNCTIONS
// =====================================================

// Relay is Active LOW
// LOW  = ON
// HIGH = OFF

void beltOn() {

  digitalWrite(BELT_RELAY, LOW);

  Serial.println("BELT ON");
}


void beltOff() {

  digitalWrite(BELT_RELAY, HIGH);

  Serial.println("BELT OFF");
}


// =====================================================
//                  SERVO FUNCTIONS
// =====================================================

void servoPulse(int angle) {

  int pwm = map(angle, 0, 180, 500, 2400);

  digitalWrite(SERVO_PIN, HIGH);

  delayMicroseconds(pwm);

  digitalWrite(SERVO_PIN, LOW);

  delay(20);
}


void foodOpen() {

  Serial.println("FOOD OPEN");

  for (int angle = 40; angle <= 64; angle += 3) {

    servoPulse(angle);
  }
}


void foodClose() {

  Serial.println("FOOD CLOSE");

  for (int angle = 64; angle >= 40; angle -= 3) {

    servoPulse(angle);
  }
}


// =====================================================
//                  ULTRASONIC
// =====================================================

void readUltrasonic() {

  digitalWrite(TRIG_PIN, LOW);

  delayMicroseconds(2);

  digitalWrite(TRIG_PIN, HIGH);

  delayMicroseconds(10);

  digitalWrite(TRIG_PIN, LOW);


  duration = pulseIn(ECHO_PIN, HIGH, 5000);


  if (duration == 0) {

    distance = 14;

  } else {

    distance = duration * 0.034 / 2;
  }


  wastePercent = map(distance, 2, 14, 100, 0);

  wastePercent = constrain(wastePercent, 0, 100);
}


// =====================================================
//                  TURBIDITY
// =====================================================

void readTurbidity() {

  int sensorValue = analogRead(TURBIDITY_PIN);

  turbidityPercent =
    map(sensorValue, 0, 4095, 100, 0);

  turbidityPercent =
    constrain(turbidityPercent, 0, 100);
}


// =====================================================
//              SEND COMMAND AND RETURN HOME
// =====================================================

void goHome() {

  server.sendHeader("Location", "/");

  server.send(303, "text/plain", "");
}

// =====================================================
//                  SETUP ROUTES
// =====================================================

void setupRoutes() {

  // Home
  server.on("/", handleRoot);


  // Wheel motors

  server.on("/forward", []() {

    forwardMotor();

    Serial.println("FORWARD");

    goHome();
  });


  server.on("/reverse", []() {

    backwardMotor();

    Serial.println("REVERSE");

    goHome();
  });


  server.on("/left", []() {

    leftMotor();

    Serial.println("LEFT");

    goHome();
  });


  server.on("/right", []() {

    rightMotor();

    Serial.println("RIGHT");

    goHome();
  });


  server.on("/stop", []() {

    stopMotor();

    Serial.println("STOP");

    goHome();
  });


  // Conveyor belt

  server.on("/belt/on", []() {

    beltOn();

    goHome();
  });


  server.on("/belt/off", []() {

    beltOff();

    goHome();
  });


  // Fish feeder

  server.on("/food/open", []() {

    foodOpen();

    goHome();
  });


  server.on("/food/close", []() {

    foodClose();

    goHome();
  });

}


// =====================================================
//                       SETUP
// =====================================================

void setup() {

  Serial.begin(115200);


  // Wheel motors

  pinMode(IN1, OUTPUT);

  pinMode(IN2, OUTPUT);

  pinMode(IN3, OUTPUT);

  pinMode(IN4, OUTPUT);


  // IMPORTANT:
  // Wheel motors OFF at startup

  stopMotor();


  // Conveyor relay

  pinMode(BELT_RELAY, OUTPUT);


  // IMPORTANT:
  // Conveyor OFF at startup

  digitalWrite(BELT_RELAY, HIGH);


  // Servo

  pinMode(SERVO_PIN, OUTPUT);


  // Ultrasonic

  pinMode(TRIG_PIN, OUTPUT);

  pinMode(ECHO_PIN, INPUT);


  // Turbidity

  pinMode(TURBIDITY_PIN, INPUT);


  // WiFi

  Serial.println();

  Serial.println("Connecting to WiFi...");

  WiFi.begin(ssid, password);


  while (WiFi.status() != WL_CONNECTED) {

    delay(500);

    Serial.print(".");

  }


  Serial.println();

  Serial.println("WiFi Connected!");


  Serial.print("ESP32 IP Address: ");

  Serial.println(WiFi.localIP());


  // Start server

  setupRoutes();

  server.begin();


  Serial.println("Web Server Started!");

}


// =====================================================
//                       LOOP
// =====================================================

void loop() {

  server.handleClient();

}
